4. Agregasi dasar: Jumlah kunjungan per poli

6. Bukti Dependensi Fungsional (0 pelanggaran di seluruh tabel)
SQL: SELECT 'id_poli -> nama_poli,lantai' AS fd, COUNT(*) AS pelanggaran FROM (SELECT id_poli FROM poli GROUP BY id_poli HAVING COUNT(DISTINCT nama_poli)>1 OR COUNT(DISTINCT lantai)>1) x UNION ALL SELECT 'no_sip -> id_dokter', COUNT(*) FROM (SELECT no_sip FROM dokter GROUP BY no_sip HAVING COUNT(DISTINCT id_dokter)>1) x UNION ALL SELECT 'nik -> id_pasien', COUNT(*) FROM (SELECT nik FROM pasien GROUP BY nik HAVING COUNT(DISTINCT id_pasien)>1) x UNION ALL SELECT '(id_dokter,tanggal,jam_mulai) -> id_jadwal', COUNT(*) FROM (SELECT id_dokter,tanggal,jam_mulai FROM jadwal_praktik GROUP BY 1,2,3 HAVING COUNT(DISTINCT id_jadwal)>1) x UNION ALL SELECT '(id_jadwal,no_antrean) -> id_kunjungan', COUNT(*) FROM (SELECT id_jadwal,no_antrean FROM kunjungan GROUP BY 1,2 HAVING COUNT(DISTINCT id_kunjungan)>1) x UNION ALL SELECT '(id_pasien,id_jadwal) -> id_kunjungan', COUNT(*) FROM (SELECT id_pasien,id_jadwal FROM kunjungan GROUP BY 1,2 HAVING COUNT(DISTINCT id_kunjungan)>1) x UNION ALL SELECT 'id_kunjungan -> id_rekam', COUNT(*) FROM (SELECT id_kunjungan FROM rekam_medis GROUP BY 1 HAVING COUNT(DISTINCT id_rekam)>1) x UNION ALL SELECT '(id_kunjungan,id_tindakan) -> jumlah', COUNT(*) FROM (SELECT id_kunjungan,id_tindakan FROM detail_tindakan GROUP BY 1,2 HAVING COUNT(*)>1) x UNION ALL SELECT '(id_rekam,id_obat) -> jumlah', COUNT(*) FROM (SELECT id_rekam,id_obat FROM detail_resep GROUP BY 1,2 HAVING COUNT(*)>1) x UNION ALL SELECT 'id_kunjungan -> id_pembayaran', COUNT(*) FROM (SELECT id_kunjungan FROM pembayaran GROUP BY 1 HAVING COUNT(DISTINCT id_pembayaran)>1) x;
---------------------------------------------------------------------

7. Bukti Keutuhan Relasi (0 kunci tamu yatim, verify.sh bagian B)
SQL: SELECT COUNT(*) AS jumlah_relasi_yatim FROM (SELECT d.id_dokter FROM dokter d LEFT JOIN poli p ON p.id_poli=d.id_poli WHERE p.id_poli IS NULL UNION ALL SELECT j.id_jadwal FROM jadwal_praktik j LEFT JOIN dokter d ON d.id_dokter=j.id_dokter WHERE d.id_dokter IS NULL UNION ALL SELECT k.id_kunjungan FROM kunjungan k LEFT JOIN pasien p ON p.id_pasien=k.id_pasien WHERE p.id_pasien IS NULL UNION ALL SELECT k.id_kunjungan FROM kunjungan k LEFT JOIN jadwal_praktik j ON j.id_jadwal=k.id_jadwal WHERE j.id_jadwal IS NULL UNION ALL SELECT r.id_rekam FROM rekam_medis r LEFT JOIN kunjungan k ON k.id_kunjungan=r.id_kunjungan WHERE k.id_kunjungan IS NULL UNION ALL SELECT dt.id_detail_tindakan FROM detail_tindakan dt LEFT JOIN kunjungan k ON k.id_kunjungan=dt.id_kunjungan WHERE k.id_kunjungan IS NULL UNION ALL SELECT dt.id_detail_tindakan FROM detail_tindakan dt LEFT JOIN tindakan t ON t.id_tindakan=dt.id_tindakan WHERE t.id_tindakan IS NULL UNION ALL SELECT dr.id_detail_resep FROM detail_resep dr LEFT JOIN rekam_medis r ON r.id_rekam=dr.id_rekam WHERE r.id_rekam IS NULL UNION ALL SELECT dr.id_detail_resep FROM detail_resep dr LEFT JOIN obat o ON o.id_obat=dr.id_obat WHERE o.id_obat IS NULL UNION ALL SELECT b.id_pembayaran FROM pembayaran b LEFT JOIN kunjungan k ON k.id_kunjungan=b.id_kunjungan WHERE k.id_kunjungan IS NULL) x;
